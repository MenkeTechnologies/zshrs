# Snapshot of free `fn` names in src/ported/**.rs that have no
# matching function definition in zsh's upstream C source. Each line
# is one identifier — the test reads this file and treats anything in
# it as "exempt-for-now". Anything NOT in this file that appears as a
# free fn in src/ported/ AND has no C counterpart fails the test.
#
# Process to fix a violation: inline the body at every call site (or
# rename to match a real C function name), then remove the line from
# this file. The list shrinks as the port catches up to the freeze.
#
# To regenerate after intentional removals:
#   cargo test --test ported_fn_names_match_c -- --nocapture 2>&1 \
#     | grep "no C counterpart" \
#     | sed -E 's/.*fn ([a-zA-Z_][a-zA-Z_0-9]*).*/\1/' \
#     | sort -u > /tmp/new_violations.txt
#   then merge with the existing list, keeping the header.
#
# Lines starting with `#` are comments. Blank lines ignored.

# Rust-only architectural helpers — no C counterpart by design.
# zsh C uses the global `sigtrapped[]` / `sigfuncs[]` arrays
# directly (no accessor function); zshrs needs a singleton to own
# the OnceLock<TrapHandler>. Inlining at 18 call sites would mean
# repeating `TRAPS.get_or_init(...)` everywhere.
# zsh C forks before installing handlers, so it has no analog. Rust's
# Rayon worker pool means the signal handler may run in a forked
# child and must skip touching parent state. Used by glob.rs's parallel
# expansion path to disable post-fork worker thread access.

# Color-distance / 256-palette mapping helpers — extracted from
# nearcolor module's main lookup logic. zsh C does this inline.

# Rust-only RNG primitives — zsh C uses `random()` libc directly.
# We wrap to centralise the seed/state and provide ChaCha20Rng for
# tests. These names exist nowhere in zsh C.

# OpenSSH BSD setresuid/setresgid shim helpers — these are Rust
# wrappers around libc::setres{uid,gid}() with errno save/restore;
# zsh C has the same shim file (openssh_bsd_setres_id.c) but uses
# direct errno macros, not function-shaped helpers.

# init.rs argv-parsing / help-text / exe-resolution helpers —
# zsh C inlines this work in `parseargs()` and `init_main()`.
# Each is single-caller and < 30 lines.

# tcp.rs Rust-side helpers extracted from `bin_ztcp` (Src/Modules/
# tcp.c). C zsh inlines connect/listen/accept/etc. inside one large
# bin_ztcp dispatcher; Rust split into modular helpers for testability.

# pattern.rs / glob.rs extracted helpers (parse / match-impl / range
# compare). zsh C inlines this work in `pattern.c::patcompile`,
# `pattern.c::patmatch`, and `glob.c`'s sort comparators.

# watch.rs Rust-side utmp/utmpx readers + format helpers — zsh C
# inlines this work in `watchlog2` (Src/Modules/watch.c).

# utils.rs Rust-side stdlib wrappers and zshrs-original helpers
# kept for ergonomics — zsh C inlines these or uses macros.

# math.rs — `name[idx]OP rhs` subscript-arith string parsers. zsh C
# does not pre-parse the assignment text: its `setmathvar()` (math.c:972)
# walks the `lval` left in place by the lexer/parser, so subscripts
# fall out of the regular lexer path. zshrs sees the raw `((a[i]+=expr))`
# text and must split it before evaluating; these three parsers do that
# split. Moving to src/extensions/ once the lexer can re-emit subscript
# tokens directly.
parse_compound
parse_pre_inc
parse_assign

# math.rs save_state/restore_state — Rust-side wrappers for C

# Rust port factors them into helpers since recursive eval happens
# from two callsites (callmathfunc arg parsing, getmathparam

save_state

# rlimits.rs — Rust-only helpers in the faithful port of
# Src/Builtins/rlimits.c. C uses preprocessor / direct globals where
# Rust needs runtime helpers:
#   nlimits — wraps `RLIM_NLIMITS as usize` to abstract the per-libc
#     return type (`c_int` macOS vs `__rlimit_resource_t` Linux).
#   ensure_limits_initialized — seeds `LIMITS`/`CURRENT_LIMITS` on
#     first use; C does this inline in `init_main()` (Src/init.c:1287).
#   leak_unknown_name — `Box::leak`s the `UNKNOWN-N` placeholder so
#     `resinfo_T.name` (`&'static str`) holds a stable pointer; C
#     uses `zalloc(12)` (rlimits.c:207) which is never freed for the
#     program's lifetime either.
#   lookup_resinfo — bounds-checked + lock-acquired access to RESINFO;
#     C inlines `resinfo[lim]->...` everywhere.
nlimits
ensure_limits_initialized
leak_unknown_name
lookup_resinfo

# sched.rs — Rust-only helpers in the faithful port of
# Src/Builtins/sched.c. C inlines or uses libc primitives where Rust
# needs runtime helpers:
#   schedcmds_lock / schedcmdtimed_lock — OnceLock get-or-init for the
#     two module statics; C dereferences `schedcmds`/`schedcmdtimed`
#     directly.
#   checksched_thunk — fn() adapter for `addtimedfn`; C `checksched`
#     returns void, Rust port returns i32 so module entries can use it.
#   now_secs — wraps SystemTime to return seconds; C uses `time(NULL)`.
#   parse_time_spec — extracts the `bin_sched` time-parsing block
#     (sched.c:236-306) into its own fn; C inlines.
#   parse_leading_decimal — Rust port of the `zstrtol(s, &s, 10)`
#     pointer-advance idiom; C uses zstrtol directly.
#   midnight_today_local — Rust port of C's `localtime()`+arithmetic
#     for "epoch second of midnight today (local TZ)".
#   format_localtime — wraps chrono's strftime to mirror C's
#     `ztrftime("%a %b %e %k:%M:%S")`.
#   addtimedfn / deltimedfn — stubs for utils.c declarations pending
#     the timed-fn machinery port. Note: these names ARE in C but
#     in utils.c, not sched.c — they're stubs locally.
#   execstring — stub for the exec.c declaration; rewires to fusevm
#     bytecode VM when `checksched` is invoked from the eval loop.
schedcmds_lock
