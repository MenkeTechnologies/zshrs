# Snapshot of free `fn` names in src/ported/**.rs that have no
# matching function definition in zsh's upstream C source. Each line
# is one identifier — the test reads this file and treats anything in
# it as "exempt-for-now". Anything NOT in this file that appears as a
# free fn in src/ported/ AND has no C counterpart fails the test.
#
# Process to fix a violation: inline the body at every call site (or
# rename to match a real C function name), then remove the line from
# this file. The list shrinks as the port catches up to the freeze.
#
# To regenerate after intentional removals:
#   cargo test --test ported_fn_names_match_c -- --nocapture 2>&1 \
#     | grep "no C counterpart" \
#     | sed -E 's/.*fn ([a-zA-Z_][a-zA-Z_0-9]*).*/\1/' \
#     | sort -u > /tmp/new_violations.txt
#   then merge with the existing list, keeping the header.
#
# Lines starting with `#` are comments. Blank lines ignored.

# Rust-only architectural helpers — no C counterpart by design.
# zsh C uses the global `sigtrapped[]` / `sigfuncs[]` arrays
# directly (no accessor function); zshrs needs a singleton to own
# the OnceLock<TrapHandler>. Inlining at 18 call sites would mean
# repeating `TRAPS.get_or_init(...)` everywhere.
traps
# zsh C forks before installing handlers, so it has no analog. Rust's
# Rayon worker pool means the signal handler may run in a forked
# child and must skip touching parent state. Used by glob.rs's parallel
# expansion path to disable post-fork worker thread access.
is_forked_child
# Helper extracted from `printjob` (Src/jobs.c) status-formatting
# inline block. Used by 2 sites in jobs.rs printjob; inlining
# would double the 25-line status-decode block.
format_process_status
# Inner numeric-compare branch of zstrcmp/eltpcmp — direct port
# of src/zsh/Src/sort.c:137-172 (the `if (sortnumeric)` block).
# 100+ lines; inlining into zstrcmp would make it unwieldy.
compare_numeric

lgamma_fn

bessel_y0

# direct errno macros, not function-shaped helpers.
errno

errno_str

# init.rs argv-parsing / help-text / exe-resolution helpers —
